name: multiplication_binaire_3_bandes
init: qInit
accept: qAccept

// ==========================================
// PHASE 1 : PARSING ET COPIE DE Y
// ==========================================
// qInit : Cherche le séparateur # sur le ruban 1
qInit, 0, _, _
qInit, 0, _, _, >, -, -

qInit, 1, _, _
qInit, 1, _, _, >, -, -

qInit, #, _, _
qCopyY, _, _, _, >, -, -

// qCopyY : Transfère y du Ruban 1 vers le Ruban 2
qCopyY, 0, _, _
qCopyY, _, 0, _, >, >, -

qCopyY, 1, _, _
qCopyY, _, 1, _, >, >, -

// Fin de Y. T1 et T2 reculent pour se placer sur le dernier bit
qCopyY, _, _, _
qSetupT1, _, _, _, <, <, -

// ==========================================
// PHASE 2 : ALIGNEMENT SUR LE LSB DE X
// ==========================================
// qSetupT1 : Le R1 recule jusqu'à trouver le LSB de x (saute les _)
qSetupT1, _, 0, _
qSetupT1, _, 0, _, <, -, -

qSetupT1, _, 1, _
qSetupT1, _, 1, _, <, -, -

// LSB de x trouvé ! On lance la boucle principale
qSetupT1, 0, 0, _
qLoop, 0, 0, _, -, -, -

qSetupT1, 1, 0, _
qLoop, 1, 0, _, -, -, -

qSetupT1, 0, 1, _
qLoop, 0, 1, _, -, -, -

qSetupT1, 1, 1, _
qLoop, 1, 1, _, -, -, -


// ==========================================
// PHASE 3 : DECREMENTATION DE Y (R2 = R2 - 1)
// ==========================================
// Si R2 lit 1 -> devient 0, pas de retenue, on passe à l'addition
qLoop, 0, 1, _
qAdd0, 0, 0, _, -, >, -

qLoop, 1, 1, _
qAdd0, 1, 0, _, -, >, -

qLoop, 0, 1, 0
qAdd0, 0, 0, 0, -, >, -

qLoop, 1, 1, 0
qAdd0, 1, 0, 0, -, >, -

qLoop, 0, 1, 1
qAdd0, 0, 0, 1, -, >, -

qLoop, 1, 1, 1
qAdd0, 1, 0, 1, -, >, -

// Si R2 lit 0 -> On doit emprunter (Borrow) vers la gauche
qLoop, 0, 0, _
qLoopBorrow, 0, 1, _, -, <, -

qLoop, 1, 0, _
qLoopBorrow, 1, 1, _, -, <, -

qLoop, 0, 0, 0
qLoopBorrow, 0, 1, 0, -, <, -

qLoop, 1, 0, 0
qLoopBorrow, 1, 1, 0, -, <, -

qLoop, 0, 0, 1
qLoopBorrow, 0, 1, 1, -, <, -

qLoop, 1, 0, 1
qLoopBorrow, 1, 1, 1, -, <, -

// Gestion du Borrow (Propagation de la retenue sur Y)
// R2 lit 0 -> devient 1, continue à gauche
qLoopBorrow, 0, 0, _
qLoopBorrow, 0, 1, _, -, <, -

qLoopBorrow, 1, 0, _
qLoopBorrow, 1, 1, _, -, <, -

qLoopBorrow, 0, 0, 0
qLoopBorrow, 0, 1, 0, -, <, -

qLoopBorrow, 1, 0, 0
qLoopBorrow, 1, 1, 0, -, <, -

qLoopBorrow, 0, 0, 1
qLoopBorrow, 0, 1, 1, -, <, -

qLoopBorrow, 1, 0, 1
qLoopBorrow, 1, 1, 1, -, <, -

// R2 lit 1 -> devient 0, emprunt terminé, on rembobine Y vers la droite
qLoopBorrow, 0, 1, _
qRewindY, 0, 0, _, -, >, -

qLoopBorrow, 1, 1, _
qRewindY, 1, 0, _, -, >, -

qLoopBorrow, 0, 1, 0
qRewindY, 0, 0, 0, -, >, -

qLoopBorrow, 1, 1, 0
qRewindY, 1, 0, 0, -, >, -

qLoopBorrow, 0, 1, 1
qRewindY, 0, 0, 1, -, >, -

qLoopBorrow, 1, 1, 1
qRewindY, 1, 0, 1, -, >, -

// UNDERFLOW : Si Y cherchait un emprunt et tombe dans le vide, Y = 0. CALCUL TERMINÉ.
qLoopBorrow, 0, _, _
qAccept, 0, _, _, -, -, -

qLoopBorrow, 1, _, _
qAccept, 1, _, _, -, -, -

qLoopBorrow, 0, _, 0
qAccept, 0, _, 0, -, -, -

qLoopBorrow, 1, _, 0
qAccept, 1, _, 0, -, -, -

qLoopBorrow, 0, _, 1
qAccept, 0, _, 1, -, -, -

qLoopBorrow, 1, _, 1
qAccept, 1, _, 1, -, -, -

// Rembobinage de Y sur son LSB
qRewindY, 0, 0, _
qRewindY, 0, 0, _, -, >, -

qRewindY, 1, 0, _
qRewindY, 1, 0, _, -, >, -

qRewindY, 0, 1, _
qRewindY, 0, 1, _, -, >, -

qRewindY, 1, 1, _
qRewindY, 1, 1, _, -, >, -

qRewindY, 0, 0, 0
qRewindY, 0, 0, 0, -, >, -

qRewindY, 1, 0, 0
qRewindY, 1, 0, 0, -, >, -

qRewindY, 0, 1, 0
qRewindY, 0, 1, 0, -, >, -

qRewindY, 1, 1, 0
qRewindY, 1, 1, 0, -, >, -

qRewindY, 0, 0, 1
qRewindY, 0, 0, 1, -, >, -

qRewindY, 1, 0, 1
qRewindY, 1, 0, 1, -, >, -

qRewindY, 0, 1, 1
qRewindY, 0, 1, 1, -, >, -

qRewindY, 1, 1, 1
qRewindY, 1, 1, 1, -, >, -

// Fin du rembobinage de Y, on le parque sur _ et on lance l'addition
qRewindY, 0, _, _
qAdd0, 0, _, _, -, -, -

qRewindY, 1, _, _
qAdd0, 1, _, _, -, -, -

qRewindY, 0, _, 0
qAdd0, 0, _, 0, -, -, -

qRewindY, 1, _, 0
qAdd0, 1, _, 0, -, -, -

qRewindY, 0, _, 1
qAdd0, 0, _, 1, -, -, -

qRewindY, 1, _, 1
qAdd0, 1, _, 1, -, -, -


// ==========================================
// PHASE 4 : ADDITION (R3 = R3 + R1)
// ==========================================
// qAdd0 : Sans retenue entrante
qAdd0, 0, _, _
qAdd0, 0, _, 0, <, -, <

qAdd0, 1, _, _
qAdd0, 1, _, 1, <, -, <

qAdd0, 0, _, 0
qAdd0, 0, _, 0, <, -, <

qAdd0, 1, _, 0
qAdd0, 1, _, 1, <, -, <

qAdd0, 0, _, 1
qAdd0, 0, _, 1, <, -, <

qAdd0, 1, _, 1
qAdd1, 1, _, 0, <, -, <

// x est épuisé, mais l'accumulateur a encore des bits
qAdd0, _, _, 0
qAdd0, _, _, 0, -, -, <

qAdd0, _, _, 1
qAdd0, _, _, 1, -, -, <

// Addition terminée, on rembobine X et Acc
qAdd0, _, _, _
qRewindAdd, _, _, _, >, -, >


// qAdd1 : Avec retenue entrante (Carry = 1)
qAdd1, 0, _, _
qAdd0, 0, _, 1, <, -, <

qAdd1, 1, _, _
qAdd1, 1, _, 0, <, -, <

qAdd1, 0, _, 0
qAdd0, 0, _, 1, <, -, <

qAdd1, 1, _, 0
qAdd1, 1, _, 0, <, -, <

qAdd1, 0, _, 1
qAdd1, 0, _, 0, <, -, <

qAdd1, 1, _, 1
qAdd1, 1, _, 1, <, -, <

// x est épuisé, on propage la retenue sur l'accumulateur
qAdd1, _, _, 0
qAdd0, _, _, 1, -, -, <

qAdd1, _, _, 1
qAdd1, _, _, 0, -, -, <

// Fin d'addition avec création d'un nouveau bit (Overhang)
qAdd1, _, _, _
qAdd0, _, _, 1, -, -, <


// ==========================================
// PHASE 5 : REMBOBINAGE POST-ADDITION
// ==========================================
// qRewindAdd ramène X (R1) et Acc (R3) sur leur LSB respectif
qRewindAdd, 0, _, 0
qRewindAdd, 0, _, 0, >, -, >

qRewindAdd, 1, _, 0
qRewindAdd, 1, _, 0, >, -, >

qRewindAdd, 0, _, 1
qRewindAdd, 0, _, 1, >, -, >

qRewindAdd, 1, _, 1
qRewindAdd, 1, _, 1, >, -, >

// Cas où X est plus court que l'Accumulateur (X est déjà sur _, Acc recule)
qRewindAdd, _, _, 0
qRewindAdd, _, _, 0, -, -, >

qRewindAdd, _, _, 1
qRewindAdd, _, _, 1, -, -, >

// Cas inverse
qRewindAdd, 0, _, _
qRewindAdd, 0, _, _, >, -, -

qRewindAdd, 1, _, _
qRewindAdd, 1, _, _, >, -, -

// Tous les rubans sont de retour à droite. On fait un pas à gauche pour se placer 
// sur le LSB de X, le LSB de Y (qui était parqué), et le LSB de Acc.
qRewindAdd, _, _, _
qLoop, _, _, _, <, <, <
